clients.delete(k) }) }) } function sendDoorbell(signPubkey, requestId) { const ws = clients.get(signPubkey.toLowerCase()) if (!ws || ws.readyState !== ws.OPEN) return false ws.send(JSON.stringify({ type: 'kea_request', requestId })) return true } async function createRequest(clean) { const jws = await signJws(clean) const item = { requestId: clean.requestId, signPubkey: router.get('/kea/requests/:requestId', (req, res) => { const item = requests.get(req.params.requestId) if (!item) return res.status(404).json({ error: 'request not found' }) item.status = 'fetched' 'signed' item.result = result item.saveData = saveData res.json({ ok: true, saveData }) }) async function handleKpoolTx(msg) { const clean = await buildCleanKeaRequestFromKpoolTx({ msg, kpoolBase })
